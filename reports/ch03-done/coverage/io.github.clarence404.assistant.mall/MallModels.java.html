package io.github.clarence404.assistant.mall;

import com.fasterxml.jackson.annotation.JsonIgnoreProperties;
import java.time.LocalDateTime;
import java.util.List;

/**
 * 商城接口的请求与响应体：只声明智能客服用得到的字段，其余字段忽略（{@code ignoreUnknown}），商城加字段不影响这边。
 *
 * <p>所在层：外部系统接入（第 3 章引入），字段名与 mall-evolution 网关的对外接口一致（见它的 {@code api/openapi.json}）。
 * 金额、skuId 在商城那边按字符串输出（19 位整数超过 JS 的安全整数），这里也用 String 接收，不做计算。
 */
public final class MallModels {

    private MallModels() {}

    /**
     * {@code POST /api/auth/login} 的请求体。
     *
     * @param phone 手机号，演示账号 13800000000
     * @param password 密码，演示账号 mall123
     */
    public record LoginRequest(String phone, String password) {}

    /**
     * {@code POST /api/auth/login} 的响应：访问令牌（JWT，30 分钟）与刷新令牌。智能客服只把访问令牌交给页面，刷新令牌不往外传。
     *
     * @param accessToken 访问令牌，之后调订单接口时放在 {@code Authorization: Bearer ...} 里
     * @param refreshToken 刷新令牌，智能客服不使用
     * @param expiresIn 访问令牌有效期，秒
     */
    @JsonIgnoreProperties(ignoreUnknown = true)
    public record Tokens(String accessToken, String refreshToken, long expiresIn) {}

    /**
     * {@code GET /api/orders/{orderNo}} 的响应（商城的 OrderView），只取客服回答用得到的字段。
     *
     * @param orderNo 订单号
     * @param status 状态枚举名：PENDING_PAYMENT / PENDING_SHIPMENT / SHIPPED / COMPLETED / CANCELLED / CLOSED
     * @param totalAmount 商品总额
     * @param payAmount 实付金额
     * @param items 订单里的商品
     * @param createdAt 下单时间
     * @param paidAt 支付时间，未支付为 null
     */
    @JsonIgnoreProperties(ignoreUnknown = true)
    public record Order(
            String orderNo,
            String status,
            String totalAmount,
            String payAmount,
            List<OrderItem> items,
            LocalDateTime createdAt,
            LocalDateTime paidAt) {}

    /**
     * 订单里的一行商品。
     *
     * @param skuId 商品编号
     * @param productName 下单时的商品名（快照）
     * @param unitPrice 单价
     * @param quantity 数量
     */
    @JsonIgnoreProperties(ignoreUnknown = true)
    public record OrderItem(String skuId, String productName, String unitPrice, int quantity) {}

    /**
     * {@code GET /api/products/{skuId}} 的响应：商品详情 + 是否有货。商品详情匿名可查，不需要登录。
     *
     * @param product 商品信息
     * @param inStock 是否有货：商城的「库存展示」，10 秒缓存，只是提示，下单时以实际扣减为准
     */
    @JsonIgnoreProperties(ignoreUnknown = true)
    public record ProductPage(Product product, boolean inStock) {}

    /**
     * 商品信息。
     *
     * @param skuId 商品编号
     * @param name 商品名
     * @param price 价格
     * @param onShelf 是否在售（下架的商品还能查到，但不能下单）
     */
    @JsonIgnoreProperties(ignoreUnknown = true)
    public record Product(String skuId, String name, String price, boolean onShelf) {}

    /**
     * 商城失败时返回的 ProblemDetail（RFC 9457），只取业务码和说明。
     *
     * @param code 业务码，如 ORDER_NOT_FOUND、PRODUCT_NOT_FOUND、USER_LOGIN_FAILED
     * @param detail 说明，如「订单 xxx 不存在」
     */
    @JsonIgnoreProperties(ignoreUnknown = true)
    public record Problem(String code, String detail) {}
}
