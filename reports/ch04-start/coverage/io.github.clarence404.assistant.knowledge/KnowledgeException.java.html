package io.github.clarence404.assistant.knowledge;

import java.sql.SQLException;
import org.springframework.dao.DataAccessException;

/**
 * 知识库操作失败：已归类（{@link KnowledgeFailure}）并带一句能直接给用户看的说明（第 4 章引入）。
 *
 * <p>所在层：错误处理。知识库各处抛出，{@code error.ApiExceptionHandler} 转成 ProblemDetail；启动时自动入库遇到它只打警告。
 */
public class KnowledgeException extends RuntimeException {

    private static final int MAX_CAUSE_DEPTH = 16;

    private final KnowledgeFailure failure;

    /**
     * 创建异常。
     *
     * @param failure 分类，决定状态码和错误码
     * @param detail 给用户看的说明（ProblemDetail 的 detail），不含堆栈、密码等内部细节
     * @param cause 原始异常，只写日志；可以为 null
     */
    public KnowledgeException(KnowledgeFailure failure, String detail, Throwable cause) {
        super(detail, cause);
        this.failure = failure;
    }

    /** 失败分类。 */
    public KnowledgeFailure failure() {
        return failure;
    }

    /**
     * 把入库 / 检索时抛出的任意异常归类：已归类的原样返回；cause 链上有数据库异常（{@link DataAccessException}、{@link SQLException}）
     * 的算向量库不可用；其余算向量模型不可用。
     *
     * <p>为什么要沿 cause 链找：{@code VectorStore#add} 里先调向量模型、再写库，两种失败从同一个方法里抛出来；而且 PgVectorStore
     * 按条件删除失败时会把 SQL 异常包成 {@code IllegalStateException}，只看最外层会误判成模型的问题。
     *
     * @param error 任意异常
     * @param embeddingModel 当前向量模型名，写进说明里
     * @param location 向量库位置，写进说明里
     * @return 归好类的异常，从不返回 null
     */
    public static KnowledgeException classify(Throwable error, String embeddingModel, String location) {
        if (error instanceof KnowledgeException known) {
            return known;
        }
        Throwable current = error;
        for (int depth = 0; current != null && depth < MAX_CAUSE_DEPTH; depth++) {
            if (current instanceof DataAccessException || current instanceof SQLException) {
                return new KnowledgeException(
                        KnowledgeFailure.VECTOR_STORE_UNAVAILABLE,
                        "向量库 " + location + " 连不上或执行出错，请确认 PostgreSQL（带 pgvector 扩展）已启动、连接配置正确。",
                        error);
            }
            current = current.getCause();
        }
        return embeddingUnavailable(embeddingModel, error);
    }

    /**
     * 向量模型不可用。
     *
     * @param embeddingModel 模型名
     * @param cause 原始异常
     * @return 归好类的异常
     */
    public static KnowledgeException embeddingUnavailable(String embeddingModel, Throwable cause) {
        return new KnowledgeException(
                KnowledgeFailure.EMBEDDING_UNAVAILABLE,
                "向量模型 " + embeddingModel + " 连不上或调用出错，请确认模型服务已启动并已下载该模型（Ollama：ollama pull " + embeddingModel + "）。",
                cause);
    }
}
